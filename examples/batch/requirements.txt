# model load.
#

# this line to upgrade, and reactor.yaml's build.runtime_version to match.

# needs the pin moved to that release before it builds.
reactor-runtime==3.8.2

# PyTorch runs the network on a GPU. Optional: without it the model runs on
# NumPy, which the runtime already ships.
