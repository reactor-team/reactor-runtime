# Python dependencies installed into the model image in one pip pass, so a
# conflict between the runtime and the model's deps fails the build instead of
# surfacing at model load.
#
# reactor-runtime is the runtime itself; bump this line to upgrade. Releases are
# immutable — there is no `latest`.
reactor-runtime==3.1.2

# OpenCV powers the video effects. Must be >=4.10: the runtime ships numpy 2.x,
# and only opencv-python(-headless) 4.10+ is built against the numpy 2 ABI. The
# headless build carries no GUI/X11 linkage, which a serving image has no use for.
opencv-python-headless>=4.10
